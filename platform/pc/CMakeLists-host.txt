# platform/pc/CMakeLists-host.txt — host (PC) build sketch; GBA Makefile untouched.
#
# Usage (once SDL2 + cmake installed):
#   cmake -f platform/pc/CMakeLists-host.txt -B build-pc -DPLATFORM_PC=1
#   cmake --build build-pc
#
# Design:
#  - Host clang/gcc, C99, no arm-none-eabi, no agbcc, no ld_script*.ld.
#  - Force-include platform/pc/gba_shim.h with -DPLATFORM_PC so every game
#    translation unit sees section-attrs emptied and REG_*/VRAM shadows.
#  - Game sources: all src/*.c EXCEPT flash-asm and sound-mixer beds that need
#    real backends first (listed below). Re-add file by file as backends land.
#  - PC sources: platform/pc/cpu_pc.c + (video/input/audio/save)_pc.c when written.
#  - Assets: extracted at build time from baserom.gba via tools/gbagfx +
#    dungeonjson (never committed); runtime loads PNG/WAV, not .4bpp/.lz.
#  - Internal resolution fixed 240x160; SDL integer-scales (--scale N).
cmake_minimum_required(VERSION 3.15)
project(pmd-red-pc C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)

find_package(SDL2 REQUIRED)

file(GLOB_RECURSE GAME_SRCS
  ${CMAKE_SOURCE_DIR}/src/*.c
)
# Stage out hardware beds until their backends land:
list(FILTER GAME_SRCS EXCLUDE REGEX ".*/(crt0|rom_header|arm_funcs|m4a_1)\\..*")
list(FILTER GAME_SRCS EXCLUDE REGEX ".*/agb_flash.*\\.c$")
list(FILTER GAME_SRCS EXCLUDE REGEX ".*/m4a\\.c$")

# Dear ImGui (git submodule) + the C++ overlay backend.
set(IMGUI_DIR ${CMAKE_SOURCE_DIR}/platform/pc/imgui)
file(GLOB IMGUI_SRCS
  ${IMGUI_DIR}/imgui.cpp
  ${IMGUI_DIR}/imgui_draw.cpp
  ${IMGUI_DIR}/imgui_tables.cpp
  ${IMGUI_DIR}/imgui_widgets.cpp
  ${IMGUI_DIR}/imgui_demo.cpp
  ${IMGUI_DIR}/backends/imgui_impl_sdl2.cpp
  ${IMGUI_DIR}/backends/imgui_impl_sdlrenderer2.cpp
)

add_executable(pmd-red-pc
  ${GAME_SRCS}
  ${CMAKE_SOURCE_DIR}/platform/pc/cpu_pc.c
  ${CMAKE_SOURCE_DIR}/platform/pc/main_pc.c
  ${CMAKE_SOURCE_DIR}/platform/pc/ui_pc.cpp
  ${IMGUI_SRCS}
  # video_pc.c input_pc.c audio_pc.c save_pc.c — add as implemented
)
target_include_directories(pmd-red-pc PRIVATE
  ${CMAKE_SOURCE_DIR}/include
  ${CMAKE_SOURCE_DIR}/platform/pc
  ${IMGUI_DIR}
  ${IMGUI_DIR}/backends
)
target_compile_definitions(pmd-red-pc PRIVATE PLATFORM_PC MODERN=2)
target_compile_options(pmd-red-pc PRIVATE
  -include ${CMAKE_SOURCE_DIR}/platform/pc/gba_shim.h
  -Wall -Wno-unknown-pragmas
)
target_link_libraries(pmd-red-pc PRIVATE SDL2::SDL2)
